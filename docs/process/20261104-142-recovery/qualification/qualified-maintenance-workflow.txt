name: Lighthouse worker maintenance
on:
  schedule:
    - cron: '17,47 * * * *'
  workflow_dispatch:
    inputs:
      issue:
        description: Optional source issue; empty scans open owner issues
        required: false
        type: string
      fixture_setup:
        description: Create the disposable issue 362 integration candidate
        required: false
        type: boolean
        default: false
      recovery_pr:
        description: Draft Product Change PR to revalidate and republish after source URL formatting repair
        required: false
        type: string
      expected_head:
        description: Full current PR head required for explicit publication recovery
        required: false
        type: string
permissions:
  contents: read
env:
  LIGHTHOUSE_MAINTENANCE_ENABLED: 'true'
  LIGHTHOUSE_RECOVERY_ENABLED: ${{ vars.LIGHTHOUSE_RECOVERY_ENABLED }}
  LIGHTHOUSE_MAINTENANCE_ISSUES: ${{ vars.LIGHTHOUSE_MAINTENANCE_ISSUES }}
jobs:
  fixture-setup:
    if: github.event_name == 'workflow_dispatch' && inputs.fixture_setup && inputs.issue == '362' && github.actor == github.repository_owner
    runs-on: ubuntu-24.04
    permissions:
      contents: write
      issues: read
      pull-requests: write
      actions: write
    steps:
      - uses: actions/checkout@v5
        with:
          persist-credentials: false
      - uses: actions/setup-node@v5
        with:
          node-version: 24
      - name: Create only the issue 362 documentation fixture
        env:
          GH_TOKEN: ${{ github.token }}
        run: node scripts/142-fixture-setup.mjs
      - uses: actions/upload-artifact@v7
        if: always()
        with:
          name: lighthouse-142-fixture
          path: 142-fixture-output
          if-no-files-found: error
  recover-product-publication:
    if: github.event_name == 'workflow_dispatch' && inputs.recovery_pr != '' && github.actor == github.repository_owner
    concurrency:
      group: lighthouse-issue-${{ inputs.issue }}
      cancel-in-progress: false
      queue: max
    runs-on: ubuntu-24.04
    permissions:
      contents: write
      pull-requests: write
    steps:
      - uses: actions/checkout@v5
        with:
          persist-credentials: false
      - uses: actions/checkout@v5
        with:
          ref: ${{ github.event.repository.default_branch }}
          path: source
          persist-credentials: false
      - uses: actions/setup-node@v5
        with:
          node-version: 24
      - run: npm ci --ignore-scripts
      - run: npm install --no-save --package-lock=false --ignore-scripts @prodshape/integration-openspec@0.6.6
      - run: npm ci --ignore-scripts
        working-directory: source
      - run: npx --no-install puppeteer browsers install chrome
        working-directory: source
      - run: node --test src/tests/product-change-files.test.mjs src/tests/product-change-publication.test.mjs
      - name: Revalidate editorial repair and republish for a new product decision
        env:
          GH_TOKEN: ${{ github.token }}
          RECOVERY_PR: ${{ inputs.recovery_pr }}
          RECOVERY_HEAD: ${{ inputs.expected_head }}
        run: node scripts/recover-product-publication.mjs
      - uses: actions/upload-artifact@v7
        with:
          name: product-publication-recovery-${{ inputs.recovery_pr }}
          path: recovery-output
          if-no-files-found: error
  scan:
    if: github.event_name == 'workflow_dispatch' && inputs.issue != '' && !inputs.fixture_setup && inputs.recovery_pr == '' && vars.LIGHTHOUSE_AUTOMATION_ENABLED == 'true' && github.actor == github.repository_owner
    runs-on: ubuntu-24.04
    permissions:
      contents: read
      actions: read
      issues: read
    outputs:
      issues: ${{ steps.scan.outputs.issues }}
    steps:
      - uses: actions/checkout@v5
        with:
          repository: juangcarmona/lighthouse
          ref: da290f969704ce1a93d58bc8d9da95f1d9045d98
          persist-credentials: false
      - uses: actions/setup-node@v5
        with:
          node-version: 24
      - id: scan
        env:
          GH_TOKEN: ${{ github.token }}
        run: node src/packages/sdlc/src/worker-maintenance.mjs scan
  observe:
    needs: scan
    if: needs.scan.outputs.issues != '[]'
    strategy:
      fail-fast: false
      max-parallel: 5
      matrix:
        issue: ${{ fromJSON(needs.scan.outputs.issues) }}
    runs-on: ubuntu-24.04
    permissions:
      contents: read
      actions: read
      checks: read
      issues: read
      pull-requests: read
    env:
      LIGHTHOUSE_ISSUE: ${{ matrix.issue }}
    steps:
      - uses: actions/checkout@v5
        with:
          repository: juangcarmona/lighthouse
          ref: da290f969704ce1a93d58bc8d9da95f1d9045d98
          persist-credentials: false
      - uses: actions/setup-node@v5
        with:
          node-version: 24
      - name: Observe workers without changing issue state
        env:
          GH_TOKEN: ${{ github.token }}
        run: node src/packages/sdlc/src/worker-maintenance.mjs observe
      - name: Preserve observations
        if: always()
        uses: actions/upload-artifact@v7
        with:
          name: lighthouse-observation-${{ matrix.issue }}
          path: ${{ runner.temp }}/lighthouse-maintenance-${{ github.run_id }}-${{ matrix.issue }}/observations.json
          retention-days: 14
          if-no-files-found: error
  reconcile:
    needs: scan
    if: needs.scan.outputs.issues != '[]'
    strategy:
      fail-fast: false
      max-parallel: 5
      matrix:
        issue: ${{ fromJSON(needs.scan.outputs.issues) }}
    concurrency:
      group: lighthouse-issue-${{ matrix.issue }}
      cancel-in-progress: false
      queue: max
    runs-on: ubuntu-24.04
    permissions:
      contents: write
      actions: write
      checks: read
      issues: write
      pull-requests: write
    env:
      LIGHTHOUSE_ISSUE: ${{ matrix.issue }}
    steps:
      - uses: actions/checkout@v5
        with:
          repository: juangcarmona/lighthouse
          ref: da290f969704ce1a93d58bc8d9da95f1d9045d98
          persist-credentials: false
      - uses: actions/setup-node@v5
        with:
          node-version: 24
      - name: Reconcile and record one bounded retry
        id: request
        env:
          GH_TOKEN: ${{ github.token }}
        run: node src/packages/sdlc/src/worker-maintenance.mjs prepare
      - name: Preserve request before dispatch
        if: steps.request.outputs.send == 'true'
        uses: actions/upload-artifact@v7
        with:
          name: ${{ steps.request.outputs.artifact }}
          path: ${{ runner.temp }}/lighthouse-maintenance-${{ github.run_id }}-${{ matrix.issue }}/dispatch.json
          retention-days: 14
          if-no-files-found: error
      - name: Send the recorded retry once
        if: steps.request.outputs.send == 'true'
        env:
          GH_TOKEN: ${{ github.token }}
          LIGHTHOUSE_DISPATCH_COMMENT: ${{ steps.request.outputs.comment }}
        run: node src/packages/sdlc/src/worker-maintenance.mjs send
      - name: Preserve verification recovery evidence
        if: always()
        uses: actions/upload-artifact@v7
        with:
          name: lighthouse-recovery-effects-${{ matrix.issue }}-${{ github.run_attempt }}
          path: |
            ${{ runner.temp }}/lighthouse-maintenance-${{ github.run_id }}-${{ matrix.issue }}/ci-recovery.json
            ${{ runner.temp }}/lighthouse-maintenance-${{ github.run_id }}-${{ matrix.issue }}/completion-recovery.json
          retention-days: 14
          if-no-files-found: ignore
      - name: Resume the recovered completion through the router
        env:
          GH_TOKEN: ${{ github.token }}
        run: node src/packages/sdlc/src/worker-maintenance.mjs signal-completion
      - name: Preserve observations
        if: always()
        uses: actions/upload-artifact@v7
        with:
          name: lighthouse-maintenance-${{ matrix.issue }}
          path: ${{ runner.temp }}/lighthouse-maintenance-${{ github.run_id }}-${{ matrix.issue }}/observations.json
          retention-days: 14
          if-no-files-found: error
  report:
    needs: [scan, reconcile]
    if: always() && needs.scan.result == 'success' && needs.scan.outputs.issues != '[]' && vars.LIGHTHOUSE_REPORT_INDEX_ISSUE != ''
    concurrency:
      group: lighthouse-worker-reports
      cancel-in-progress: false
      queue: max
    runs-on: ubuntu-24.04
    permissions:
      contents: read
      actions: read
      issues: write
    env:
      LIGHTHOUSE_REPORT_INDEX_ISSUE: ${{ vars.LIGHTHOUSE_REPORT_INDEX_ISSUE }}
      LIGHTHOUSE_OBSERVED_ISSUES: ${{ needs.scan.outputs.issues }}
    steps:
      - uses: actions/checkout@v5
        with:
          repository: juangcarmona/lighthouse
          ref: da290f969704ce1a93d58bc8d9da95f1d9045d98
          persist-credentials: false
      - uses: actions/setup-node@v5
        with:
          node-version: 24
      - uses: actions/download-artifact@v8
        with:
          pattern: lighthouse-maintenance-*
          path: ${{ runner.temp }}/worker-observations
      - name: Reconcile and update worker error reports
        env:
          GH_TOKEN: ${{ github.token }}
        run: node src/packages/sdlc/src/worker-maintenance.mjs report
  report-stalled:
    needs: [scan, observe]
    if: always() && needs.scan.result == 'success' && needs.scan.outputs.issues != '[]' && vars.LIGHTHOUSE_REPORT_INDEX_ISSUE != ''
    concurrency:
      group: lighthouse-worker-reports
      cancel-in-progress: false
      queue: max
    runs-on: ubuntu-24.04
    permissions:
      contents: read
      actions: read
      issues: write
    env:
      LIGHTHOUSE_REPORT_INDEX_ISSUE: ${{ vars.LIGHTHOUSE_REPORT_INDEX_ISSUE }}
      LIGHTHOUSE_OBSERVED_ISSUES: ${{ needs.scan.outputs.issues }}
    steps:
      - uses: actions/checkout@v5
        with:
          repository: juangcarmona/lighthouse
          ref: da290f969704ce1a93d58bc8d9da95f1d9045d98
          persist-credentials: false
      - uses: actions/setup-node@v5
        with:
          node-version: 24
      - uses: actions/download-artifact@v8
        with:
          pattern: lighthouse-observation-*
          path: ${{ runner.temp }}/worker-observations
      - name: Reconcile and update worker error reports
        env:
          GH_TOKEN: ${{ github.token }}
        run: node src/packages/sdlc/src/worker-maintenance.mjs report
